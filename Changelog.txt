Mahdolliset arvot telegram_send_mode:

"auto" = lähetä kuva+caption jos Image-URL löytyy, muuten teksti (nykyinen käytös)
"text" = aina sendMessage (ei koskaan sendPhoto)
"photo" = yritä aina sendPhoto jos image löytyy, muuten fallback tekstiin
"photo_only" = lähetä vain jos image löytyy, muuten skip (ei fallbackia)
